// 3. عرض منتجات التصنيف المختار (نسخة محسنة) function loadDeals(category, btn) { document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active')); btn.classList.add('active'); // مسح الحاوي قبل التحميل const container = document.getElementById('dealsContainer'); container.innerHTML = '

جاري جلب أفضل العروض...

'; // استخدام once لضمان جلب البيانات مرة واحدة عند الضغط ومنع التداخل db.ref('deals/' + category).once('value', (snapshot) => { const deals = snapshot.val(); container.innerHTML = ''; if (!deals) { container.innerHTML = '

لا توجد عروض حالياً في هذا القسم.

'; return; } for (let key in deals) { const item = deals[key]; container.innerHTML += `

${item.name.substring(0, 40)}...

${item.price_before} ر.س
${item.price_now} ر.س ${item.discount_percentage}% خصم
⭐ ${item.rating}
مشاهدة العرض
`; } }); }
